   357→// ---------- CONTACT + MAP + ENQUIRY FORM ----------
   358→function Contact({ section, business, onCta, submitLead }: {
   359→  section: SiteSection; business: Business; onCta: (t: string) => void; submitLead: (data: { name: string; phone: string; email: string; message: string }) => Promise<string | null>;
   360→}) {
   361→  const c = section.content as { title?: string; subtitle?: string; mapUrl?: string };
   362→  const [form, setForm] = useState({ name: "", phone: "", email: "", message: "" });
   363→  const [sending, setSending] = useState(false);
   364→  const [done, setDone] = useState(false);
   365→  const [error, setError] = useState<string | null>(null);
   366→
   367→  async function handleSubmit(e: React.FormEvent) {
   368→    e.preventDefault();
   369→    setError(null);
   370→    if (!form.name.trim() || !form.phone.trim()) {
   371→      setError("Name and phone number are required.");
   372→      return;
   373→    }
   374→    setSending(true);
   375→    const err = await submitLead(form);
   376→    setSending(false);
   377→    if (err) { setError(err); return; }
   378→    setDone(true);
   379→  }
   380→
   381→  const waNumber = (business.whatsapp || business.phone || "").replace(/[^\d]/g, "");
   382→  const mapsHref = business.mapsUrl || c.mapUrl || `https://maps.google.com/?q=${encodeURIComponent(`${business.address} ${business.city} ${business.pincode}`)}`;
   383→
   384→  return (
   385→    <section id="contact" className="py-16 md:py-20">
   386→      <div className="mx-auto max-w-6xl px-4 sm:px-6">
   387→        <SectionTitle title={c.title || "Contact Us"} subtitle={c.subtitle} />
   388→        <div className="grid gap-8 md:grid-cols-2">
   389→          {/* Info + map */}
   390→          <div className="space-y-5">

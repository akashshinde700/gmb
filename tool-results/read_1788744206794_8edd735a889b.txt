   407→  if (!upi && !customQr) return null; // nothing configured yet
   408→
   409→  const payLink = upi ? upiDeepLink(upi, business.name, "Website payment") : "";
   410→
   411→  async function copyUpi() {
   412→    if (!upi) return;
   413→    try {
   414→      await navigator.clipboard.writeText(upi);
   415→      setCopied(true);
   416→      setTimeout(() => setCopied(false), 1800);
   417→    } catch { /* clipboard unavailable */ }
   418→  }
   419→
   420→  return (
   421→    <section id="payment" className="bg-[var(--brand-surface)] py-16 md:py-20">
   422→      <div className="mx-auto max-w-4xl px-4 sm:px-6">

   425→          {/* QR */}
   426→          <div className="shrink-0 rounded-[var(--brand-radius-lg)] border-2 border-[var(--brand-primary)]/20 bg-white p-3 shadow-sm">
   427→            {customQr ? (
   428→               
   429→              <img src={customQr} alt={`${business.name} payment QR code`} className="h-44 w-44 object-contain" />
   430→            ) : (
   431→              <QRCode value={payLink} size={176} bgColor="#ffffff" fgColor="var(--brand-secondary)" />
   432→            )}
   433→            <p className="mt-2 text-center text-[10px] font-semibold uppercase tracking-widest text-[var(--brand-muted)]">
   434→              UPI · QR Code
   435→            </p>
   436→          </div>
   437→          {/* Details */}
   438→          <div className="flex-1 text-center sm:text-left">
   439→            <p className="text-sm text-[var(--brand-muted)]">Pay to</p>
   440→            <p className="text-lg font-bold text-[var(--brand-secondary)]">{business.name}</p>
   441→            {upi && (
   442→              <button
   443→                type="button"
   444→                onClick={() => void copyUpi()}
   445→                className="mt-3 inline-flex items-center gap-2 rounded-[var(--brand-radius)] border border-[var(--brand-border)] bg-[var(--brand-surface)] px-3 py-2 text-sm font-semibold text-[var(--brand-body)] transition hover:border-[var(--brand-primary)]"
   446→                aria-label="Copy UPI ID"
   447→              >
   448→                <QrIcon className="h-4 w-4 text-[var(--brand-primary)]" />
   449→                <span className="font-mono">{upi}</span>
   450→                {copied ? <Check className="h-4 w-4 text-emerald-600" /> : <Copy className="h-4 w-4 text-[var(--brand-muted)]" />}
   451→              </button>
   452→            )}
   453→            {payLink && (
   454→              <div className="mt-4">
   455→                <a
   456→                  href={payLink}
   457→                  onClick={() => onCta("CTA_CALL")}
   458→                  className="inline-flex items-center justify-center gap-2 rounded-[var(--brand-radius)] bg-[var(--brand-primary)] px-6 py-3 text-sm font-semibold text-white shadow transition hover:brightness-110 active:scale-95"
   459→                >
   460→                  <QrIcon className="h-4 w-4" /> Pay via UPI app
   461→                </a>
   462→                <p className="mt-2 text-[11px] text-[var(--brand-muted)]">GPay · PhonePe · Paytm · BHIM — opens your UPI app</p>
   463→              </div>
   464→            )}
   465→            {c.note && <p className="mt-4 text-xs leading-relaxed text-[var(--brand-muted)]">{c.note}</p>}
   466→          </div>
   467→        </div>
   468→      </div>
   469→    </section>

   474→function Hours({ section, business }: { section: SiteSection; business: Business }) {
   475→  const c = section.content as { title?: string };
   476→  const days = business.hours && Object.keys(business.hours).length
   477→    ? business.hours
   478→    : { Monday: "9:00 AM – 7:00 PM", Tuesday: "9:00 AM – 7:00 PM", Wednesday: "9:00 AM – 7:00 PM", Thursday: "9:00 AM – 7:00 PM", Friday: "9:00 AM – 7:00 PM", Saturday: "9:00 AM – 7:00 PM", Sunday: "Closed" };
   479→  const today = new Date().toLocaleDateString("en-IN", { weekday: "long" });
   480→  return (
   481→    <section className="bg-[var(--brand-surface)] py-16 md:py-20">
   482→      <div className="mx-auto max-w-2xl px-4 sm:px-6">
   483→        <SectionTitle title={c.title || "Business Hours"} />
   484→        <div className="rounded-[var(--brand-radius-lg)] bg-white p-6 ring-1 ring-[var(--brand-border)] shadow-sm">
   485→          {Object.entries(days).map(([day, time]) => (
   486→            <div key={day} className={`flex items-center justify-between border-b border-[var(--brand-border)] py-2.5 last:border-0 text-sm ${day === today ? "font-bold text-[var(--brand-primary)]" : "text-[var(--brand-body)]"}`}>
   487→              <span className="inline-flex items-center gap-2"><Clock className="h-4 w-4 text-[var(--brand-primary)]" />{day}{day === today && " (Today)"}</span>
   488→              <span>{time as string}</span>
   489→            </div>
   490→          ))}
   491→        </div>
   492→      </div>
   493→    </section>
   494→  );
   495→}
